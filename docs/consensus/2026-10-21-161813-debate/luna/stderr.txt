## Decisoes

- **DEC-001 (P-1, como o adaptador bloqueia e recupera a resposta):** `spawn` com `--shared-sandbox --prompt-file <temporario>` (prompt inteiro em arquivo, sem limite de argv nem problema de aspas; o teto de 262144 bytes e da CLI, e o adaptador repassa a mensagem dela com exit 1), depois polling de `room-cli thread output ID --all` a cada `--intervalo` segundos (padrao 10) ate `busy` falso. `state` ocioso com `reply`: imprime a resposta integral mais uma quebra de linha e sai 0. `error` preenchido ou estado de falha ou parada: stderr com o erro do filho, stdout vazio, exit 2. O adaptador nao tem timeout proprio: quem limita e o `--tempo-maximo` do loop. Filho que pergunta em vez de responder sai 0 com a pergunta no stdout; o adaptador nao distingue, e o loop ja manda gravar a pergunta em `.loop-pergunta`. def escrever_bytes_atomico(caminho, dados):
    """Temporario no mesmo diretorio, fsync e os.replace. Limpa o temporario."""
    caminho = Path(caminho)
    if isinstance(dados, str):
        dados = dados.encode("utf-8")
    temporario = caminho.parent / (".%s.%s.tmp" % (caminho.name, os.getpid()))
    try:
        with temporario.open("wb") as arquivo:
            arquivo.write(dados)
            arquivo.flush()
            os.fsync(arquivo.fileno())
        os.replace(str(temporario), str(caminho))
    finally:
        if temporario.exists():
            temporario.unlink()

    ) % (participante, copia, "" if rodada_n == 1 else "desta rodada ")
    if rodada_n == 1:

    partes = [

        partes.append(texto)
        partes.append("\n" if not texto.endswith("\n") else "")
        partes.append("----- Fim de %s -----\n" % rotulo)
    return "".join(partes)


def executa(comando, prompt, copia):
    inicio = agora()
    try:
        processo = subprocess.Popen(
            comando + [prompt],
            cwd=str(copia),
            stdin=subprocess.DEVNULL,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
        )
    except OSError as erro:
        return inicio, agora(), 127, b"", str(erro).encode("utf-8", "replace")
    with _processos_lock:
        _processos_vivos.append(processo)
    if _interrompendo.is_set():
        try:
            processo.terminate()
        except OSError:
            pass
    stdout, stderr = processo.communicate()
    with _processos_lock:
        if processo in _processos_vivos:
            _processos_vivos.remove(processo)
    return inicio, agora(), processo.returncode, stdout, stderr


def ao_sinal(signum, _frame):
    _interrompendo.set()
    with _processos_lock:
        alvos = list(_processos_vivos)
    for processo in alvos:
        try:
            processo.terminate()
        except OSError:
            pass
    # SIGINT: nao deixa o KeyboardInterrupt default furar o finally.
    if signum == signal.SIGINT:
        return

def id_inseguro(identificador):
    if not identificador or identificador in IDS_RESERVADOS:
        return True
    return "/" in identificador or "\\" in identificador or os.sep in identificador


def rodadas_nao_concluidas(projeto):
    """Run-ids em docs/consensus/ cujo manifesto nao esta `concluida`, mais antigos primeiro."""
    base = Path(projeto) / "docs" / "consensus"
    saida = []
    if not base.is_dir():
        return saida
    for pasta in sorted(base.iterdir()):
        try:
            dados = json.loads((pasta / "manifesto.json").read_text(encoding="utf-8"))
        except (OSError, ValueError):
            continue
        if isinstance(dados, dict) and dados.get("status") != "concluida":
            saida.append(pasta.name)
    return saida


def adquirir_lock(projeto, run_id=None):
    lock_dir = Path(projeto) / LOCK_NOME
    try:
        os.mkdir(str(lock_dir))
    except FileExistsError:
        print("[ERRO] outra rodada ja esta em andamento neste projeto: lock em %s."
              % lock_dir, file=sys.stderr)
        print("Se uma rodada anterior terminou de forma inesperada e voce confirmou que",
              file=sys.stderr)
        print("nao ha rodada ativa, limpe o lock orfao com: rmdir \"%s\"" % lock_dir,
              file=sys.stderr)
        # Retomada e palavra humana: a mensagem so mostra o comando, nunca o executa.
        for pendente in ([run_id] if run_id else rodadas_nao_concluidas(projeto)[-3:]):
            print("Para continuar a rodada %s depois disso: %s"
                  % (pendente, comando_de_retomada(projeto, pendente)), file=sys.stderr)
        return None
    except OSError as erro:
        print("[ERRO] nao foi possivel criar o lock da rodada em %s: %s"
              % (lock_dir, erro), file=sys.stderr)
        return None
    return lock_dir


def soltar_lock(lock_dir):
    if lock_dir is None:
        return
    try:
        os.rmdir(str(lock_dir))
    except OSError:
        pass


def citar(texto):
    """Toda linha vira citacao Markdown; U+2014 vira hifen. Devolve (bloco, n)."""
    if texto is None:
        texto = ""
    n = texto.count(TRAVESSAO)
    texto = texto.replace(TRAVESSAO, "-")
    linhas = texto.splitlines()
    if not linhas:
        return "> \n", n
    return "\n".join("> " + linha for linha in linhas) + "\n", n


def montar_minuta(titulo, dono, run_id, pergunta, participantes, data, rodada_n=1, pendente=""):
    ids = [p["id"] for p in participantes]
    falhos = [p["id"] for p in participantes if p.get("resultado") != "posicao"]
    falhas_txt = ", ".join(falhos) if falhos else "nenhuma"
    manifesto_rel = "docs/consensus/%s/manifesto.json" % run_id
    partes = [
        "## %s - %s" % (data, titulo),
        "",
        "**Status:** aberto",
        "",
        "**Proximo passo:** %s le as posicoes e preenche as secoes de julgamento" % dono,
        "",
        "**Metodo:** pareceres-independentes",
        "",
        "**Exposicao previa a outras posicoes:** %s" % ("nao" if rodada_n == 1 else "sim"),
        "",
        "**Rodada:** %d de %d" % (rodada_n, rodada_n),
        "",
    ]
    if pendente:
        partes.extend(["**Pendente da rodada anterior:** %s" % pendente, ""])
    partes += [
        "**Origem:** automacao",
        "",
        "**Manifesto:** %s" % manifesto_rel,
        "",
        "**Participantes:** %s" % ", ".join(ids),
        "",
        "**Falhas:** %s" % falhas_txt,
        "",
        "### Contexto",
        "",
        "- Modo: debate",
        "- Rodada: %d de %d" % (rodada_n, rodada_n),
        "- N: %s" % len(ids),
        "- Data: %s" % data,
        "- Manifesto: %s" % manifesto_rel,
        "",
        "### Pergunta Ou Decisao",
        "",
        citar(pergunta)[0].rstrip("\n"),
        "",
    ]
    for p in participantes:
        if p.get("resultado") == "posicao":
            partes.extend(secao_de_posicao("Posicao De", p["id"], p.get("_stdout_texto", "")) + [""])
        else:
            partes.extend(secao_de_falha(run_id, p) + [""])
    for secao in (
        "Pontos De Acordo",
        "Riscos E Tradeoffs",
        "Consenso Final",
        "Decisao Para Registrar Em DECISIONS.md",
    ):
        partes.extend(["### %s" % secao, "", "- (A preencher.)", ""])
    return "\n".join(partes).rstrip() + "\n"


def gravar_minuta(projeto, minuta):
    """Acrescenta a minuta no fim, sem alterar o prefixo. Devolve hashes e o texto."""
    caminho = Path(projeto) / "docs" / "CONSENSUS.md"
    antes = caminho.read_text(encoding="utf-8")
    sha_antes = soma_texto(antes)
    adendo = ""
    if antes and not antes.endswith("\n"):
        adendo += "\n"
    adendo += "\n" + minuta.lstrip("\n")
    depois = antes + adendo
    escrever_bytes_atomico(caminho, depois.encode("utf-8"))
    return sha_antes, soma(caminho), "docs/CONSENSUS.md"

    """Cria docs/consensus/<instante>-<sufixo>, com -2, -3 se a pasta ja existe.

def com_lock_e_sinais(projeto, funcao, run_id=None):
    """Lock exclusivo por projeto e tratamento de SIGINT/SIGTERM em volta de `funcao`."""
    lock_dir = adquirir_lock(projeto, run_id)
    if lock_dir is None:
        return 5
    anteriores = {
        signal.SIGINT: signal.getsignal(signal.SIGINT),
        signal.SIGTERM: signal.getsignal(signal.SIGTERM),
    }
    _interrompendo.clear()
    with _processos_lock:
        _processos_vivos.clear()
    try:
        signal.signal(signal.SIGINT, ao_sinal)
        signal.signal(signal.SIGTERM, ao_sinal)
        return funcao()
    finally:
        for sig, handler in anteriores.items():
            try:
                signal.signal(sig, handler)
            except (OSError, ValueError):
                pass
        soltar_lock(lock_dir)


def rodar_participantes(projeto, rodada, run_id, manifesto, trabalhos_cmd, prompt_de, reter):
    """Primitivo comum a debate e achado: uma copia isolada por participante, execucao
    em paralelo, bruto e manifesto gravados assim que cada um encerra, varredura de
    segredo antes de gravar. `reter` esconde as entradas anteriores de CONSENSUS.md na
    copia (debate); achado a deixa inteira, porque a entrada sob revisao e o objeto
    (DEC-003). Devolve {id: stdout} de quem terminou sem reter por segredo."""
    stdout_por_id = {}
    temporarios = []
    trabalhos = []
    try:
        for identificador, comando in trabalhos_cmd:
            if _interrompendo.is_set():
                break
            raiz_temporaria = Path(tempfile.mkdtemp(prefix="consenso-"))
            temporarios.append(raiz_temporaria)
            copia = raiz_temporaria / "projeto"
            shutil.copytree(str(projeto), str(copia), ignore=ignorar_copia(projeto))
            if reter:
                reter_registros(copia)
            trabalhos.append((identificador, comando, str(copia), prompt_de(identificador, str(copia))))

        if trabalhos and not _interrompendo.is_set():
            with concurrent.futures.ThreadPoolExecutor(max_workers=len(trabalhos)) as executor:
                futuros = {
                    executor.submit(executa, comando, prompt, copia): (
                        identificador, comando, copia, prompt)
                    for identificador, comando, copia, prompt in trabalhos
                }
                for futuro in concurrent.futures.as_completed(futuros):
                    identificador, comando, copia, prompt = futuros[futuro]
                    inicio, fim, codigo, stdout, stderr = futuro.result()
                    if _interrompendo.is_set() and codigo != 0:
                        # Morto pelo nosso sinal: nao ha bruto a guardar. Sem registro e sem
                        # stdout.txt, a retomada chama este participante de novo.
                        continue
                    destino = rodada / identificador
                    destino.mkdir(exist_ok=True)
                    prompt_path = destino / "prompt.txt"
                    prompt_path.write_text(prompt, encoding="utf-8")
                    if tem_segredo(stdout) or tem_segredo(stderr):
                        retido = reter_bruto_fora(stdout, stderr)
                        avisar_retido(retido, projeto, run_id, identificador)
                        registro = {
                            "id": identificador,
                            "comando": comando,
                            "diretorio": copia,
                            "inicio": inicio,
                            "fim": fim,
                            "exit": codigo,
                            "prompt": "%s/prompt.txt" % identificador,
                            "sha256_prompt": soma(prompt_path),
                            "resultado": "retido-por-segredo",
                            "retido_em": str(retido),
                        }
                        manifesto["participantes"].append(registro)
                        escrever_json(rodada / "manifesto.json", manifesto)
                        continue
                    stdout_path = destino / "stdout.txt"
                    stderr_path = destino / "stderr.txt"
                    stdout_path.write_bytes(stdout)
                    stderr_path.write_bytes(stderr)
                    stdout_texto = stdout.decode("utf-8", "replace")
                    stdout_por_id[identificador] = stdout_texto
                    registro = {
                        "id": identificador,
                        "comando": comando,
                        "diretorio": copia,
                        "inicio": inicio,
                        "fim": fim,
                        "exit": codigo,
                        "prompt": "%s/prompt.txt" % identificador,
                        "sha256_prompt": soma(prompt_path),
                        "stdout": "%s/stdout.txt" % identificador,
                        "sha256_stdout": soma(stdout_path),
                        "stderr": "%s/stderr.txt" % identificador,
                        "resultado": "posicao" if codigo == 0 else "falha",
                        "travessoes_substituidos": stdout_texto.count(TRAVESSAO),
                    }
                    manifesto["participantes"].append(registro)
                    escrever_json(rodada / "manifesto.json", manifesto)
    finally:
        for temporario in temporarios:
            shutil.rmtree(str(temporario), ignore_errors=True)
    return stdout_por_id

## P-2: `--shared-sandbox` não preserva o isolamento da DEC-003

Discordo da DEC-002 como conclusão sobre o modo compartilhado. A evidência sustenta que o filho não começa na pasta de trabalho corrente do processo que o chamou: começa na pasta da thread pai. Portanto, passar o caminho da cópia no prompt, ou fazer `cd` antes de chamar `room-cli`, não prende o filho nessa cópia.

Isso quebra o isolamento de `consenso.py`. O script cria uma cópia por participante, remove `docs/consensus/` e, no debate, retém os registros anteriores de `docs/CONSENSUS.md`. Mas o adaptador chamado como agente receberia um prompt que aponta para uma cópia e abriria o filho no espaço compartilhado da thread pai. O caminho no prompt não impede o filho de ler outros arquivos disponíveis nesse espaço. Como `consenso.py` grava cada bruto assim que o participante termina, um participante posterior também poderia alcançar a posição anterior se ela estiver nesse espaço. Isso contraria tanto a DEC-003 da 0006 quanto o critério de aceite que exige isolamento verificável sem confiar no relato do agente.

O que sobra verificável com a proposta atual é uma afirmação de escopo: o adaptador não será usado nos perfis de debate e achado, e a documentação explicará o motivo. Isso não demonstra isolamento; evita alegá-lo. A alternativa verificável já existe no código: deixar `consenso.py` chamar os agentes diretamente, preservando a cópia separada por participante, a retenção dos registros no debate e a gravação dos brutos fora das cópias de execução. Para voltar a considerar Cloudroom no consenso, seria necessário um modo de execução que comprovadamente desse a cada filho um espaço distinto, com captura da resposta integral e sem acesso às posições contemporâneas. Os fatos fornecidos não mostram que a CLI ofereça isso.

## P-3: sandbox própria não serve aos três pacotes automatizados

Concordo com a DEC-003 em excluir o modo sem `--shared-sandbox` dos pacotes automatizados, mas ajustaria a explicação: pelo comportamento medido, ele não é apenas inconveniente. O adaptador não consegue obter a resposta pelo `thread output`; recebe somente a nota de que as respostas chegam como mensagens quando os turnos terminam. Logo, não consegue cumprir o contrato de CLI do wrapper, que precisa devolver a resposta pelo stdout e um exit code.

Para o consenso, há também um problema de isolamento e proveniência. A cópia privada leva o conteúdo existente, incluindo `docs/CONSENSUS.md` e `docs/consensus/`. Ela não fornece a cópia específica por participante, preparada por `consenso.py` segundo o debate ou o achado. Além disso, sem a resposta no processo, `consenso.py` não pode gravar o bruto e o manifesto quando o participante termina, como exige a DEC-009 da 0006.

Para o loop, o filho trabalha numa cópia em outra branch. O `loop.sh` verifica alterações com `find -newer` na árvore de `$PROJETO` e executa o portão ali. Alterações feitas apenas na cópia privada não satisfazem esse fluxo. A falta de resposta capturável também impede o adaptador de entregar ao loop a saída esperada.

Assim, minha recomendação por pacote é:

| Pacote | Modo recomendado |
|---|---|
| Debate | Nenhum modo Cloudroom medido. Manter os agentes diretos de `consenso.py`. |
| Achado | Nenhum modo Cloudroom medido. Manter os agentes diretos de `consenso.py`. |
| Loop | `--shared-sandbox`, condicionado a o espaço compartilhado corresponder ao projeto que o loop precisa alterar. |

Há uma ressalva importante à DEC-004 da spec 0008: o `loop.sh` aceita `--projeto DIR` e roda o comando do agente após `cd "$PROJETO"`, mas o `room-cli` medido inicia o filho na pasta da thread pai, independentemente do diretório do chamador. Portanto, não está demonstrado que o filho sempre alterará o `$PROJETO` indicado. A bancada descrita em T-103 pode validar o caso em que o projeto coincide com a pasta compartilhada da thread; não sustenta, sozinha, uma alegação para qualquer `--projeto`. Eu restringiria a documentação a esse caso demonstrado, ou exigiria que o critério real prove que o filho altera o caminho indicado pelo loop e que o `find -newer` o detecta.

Em resumo: manteria o adaptador restrito ao loop e fora do consenso, mas mudaria os fundamentos registrados. Para o consenso, `--shared-sandbox` falha no isolamento por participante; sem compartilhamento, falha na coleta mecânica da resposta e no acesso à cópia preparada para cada participante. Para o loop, o compartilhamento é o único modo plausível entre os dois, mas sua compatibilidade com `--projeto` precisa ser delimitada ao que a bancada real comprovar.
